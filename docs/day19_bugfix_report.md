4. Полезные SQL-проверки
-- Проверка текущих материалов и видимости
SELECT id, title, category_id, version, is_visible, file_path
FROM materials ORDER BY id;

-- Количество скачиваний до и после ОДНОГО скачивания
SELECT COUNT(*) AS total_downloads FROM downloads;

-- Журнал с сотрудниками, материалами и версиями
SELECT d.id, u.username, m.title, d.file_version, d.downloaded_at
FROM downloads d
JOIN users u ON u.id = d.user_id
JOIN materials m ON m.id = d.material_id
ORDER BY d.downloaded_at DESC, d.id DESC;
